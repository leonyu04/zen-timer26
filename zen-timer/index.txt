        // 上傳檔案至 Google 雲端硬碟
        async function uploadFileToDrive(accessToken, file) {
            const metadata = {
                name: 'zen_timer_backup.json',
                mimeType: 'application/json',
            };

            // 檢查雲端是否已經有舊的備份檔，有的話先刪除或覆蓋
            const searchRes = await fetch(`https://www.googleapis.com/drive/v3/files?q=name='zen_timer_backup.json' and trashed=false`, {
                headers: { Authorization: `Bearer ${accessToken}` }
            });
            const searchData = await searchRes.json();
            
            let fileId = null;
            if (searchData.files && searchData.files.length > 0) {
                fileId = searchData.files[0].id;
            }

            const form = new FormData();
            form.append('metadata', new Blob([JSON.stringify(metadata)], { type: 'application/json' }));
            form.append('file', file);

            let url = 'https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart';
            let method = 'POST';

            if (fileId) {
                url = `https://www.googleapis.com/upload/drive/v3/files/${fileId}?uploadType=multipart`;
                method = 'PATCH';
            }

            const res = await fetch(url, {
                method: method,
                headers: { Authorization: `Bearer ${accessToken}` },
                body: form,
            });
            if (!res.ok) throw new Error('Upload failed');
        }

        // 從 Google 雲端硬碟下載備份檔
        async function downloadFileFromDrive(accessToken) {
            const searchRes = await fetch(`https://www.googleapis.com/drive/v3/files?q=name='zen_timer_backup.json' and trashed=false`, {
                headers: { Authorization: `Bearer ${accessToken}` }
            });
            const searchData = await searchRes.json();
            
            if (!searchData.files || searchData.files.length === 0) {
                return null;
            }

            const fileId = searchData.files[0].id;
            const fileRes = await fetch(`https://www.googleapis.com/drive/v3/files/${fileId}?alt=media`, {
                headers: { Authorization: `Bearer ${accessToken}` }
            });
            
            if (!fileRes.ok) throw new Error('Download failed');
            return await fileRes.json();
        }

                    let streak = parseInt(localStorage.getItem('zenMeteorStreak')) || 0;
                    let prevRate = parseFloat(localStorage.getItem('zenMeteorRate')) || 750;

                    if (yesterdayFocusValue >= (THRESHOLD_SHOOTING / 60)) {
                        streak++;
                        let improvementRatio = yesterdayFocusValue / (THRESHOLD_SHOOTING / 60);
                        let jump = 64.28 * improvementRatio;

localStorage.setItem('zenTimerSeconds', timerSeconds);

            const activeSub = subjects.find(s => s.id == activeSubjectId);
            if (activeSub) activeSub.totalSeconds += 1;
            
            saveSubjects();

            const currentTotal = subjects.reduce((sum, s) => sum + s.totalSeconds, 0); const prevTotal = currentTotal - 1;
            if (prevTotal < UNLOCK_CUSTOM_THRESHOLD && currentTotal >= UNLOCK_CUSTOM_THRESHOLD) { showTextToast("🎉 恭喜解鎖銀河階段！右側控制面板已為你開啟", 6000); renderDrawer(); }

            if (timerSeconds > 0 && timerSeconds % SECONDS_PER_CYCLE === 0) {
                const cycle = Math.floor(timerSeconds / SECONDS_PER_CYCLE);
                if (cycle > highestStageIndex) { highestStageIndex = Math.min(cycle, storyLines.length); localStorage.setItem('zenHighestStage', highestStageIndex); }
            }

            const currentCycle = Math.floor(timerSeconds / SECONDS_PER_CYCLE);
            const progress = (timerSeconds % SECONDS_PER_CYCLE) / SECONDS_PER_CYCLE;
            updateEmojiBtn(currentCycle); 
            
            updateWater(s.toString().padStart(2, '0'), progress, true); updateBacklightAndNorthStar(currentCycle, true, false);
        }

        function togglePause(e) {
            if (!isTimerMode) return; 
            if (e && e.type === 'contextmenu') return; 
            
            if (isPaused && isViewingMode) { isViewingMode = false; document.getElementById('btn-view-mode').classList.remove('active'); applyClockVisibility(); }

            const currentCycle = Math.floor(timerSeconds / SECONDS_PER_CYCLE);

            if (isPaused) {
                const isGalaxyUnlocked = totalDailySecondsGlobal > UNLOCK_CUSTOM_THRESHOLD;
                if (isGalaxyUnlocked) { showTextToast('計時已暫停（已解鎖銀河階段與右側面板）', 5000); } 
                else {
                    let nextCycleDiff = SECONDS_PER_CYCLE - (timerSeconds % SECONDS_PER_CYCLE);
                    let nextStoryIndex = currentCycle === 0 ? 0 : currentCycle - 1; let nextStoryName = nextStoryIndex < storyLines.length ? storyLines[nextStoryIndex].name : '巔峰狀態';
                    let milestones = [ { name: nextStoryName, diff: nextCycleDiff } ];
                    if (timerSeconds < THRESHOLD_FIREFLIES) milestones.push({ name: '流螢模式', diff: THRESHOLD_FIREFLIES - timerSeconds });
                    else if (timerSeconds < THRESHOLD_STARS) milestones.push({ name: '星空模式', diff: THRESHOLD_STARS - timerSeconds });
                    else if (timerSeconds < THRESHOLD_SHOOTING) milestones.push({ name: '流星模式', diff: THRESHOLD_SHOOTING - timerSeconds });
                    milestones.push({ name: '銀河解鎖', diff: (UNLOCK_CUSTOM_THRESHOLD + 1) - totalDailySecondsGlobal });
                    milestones.sort((a, b) => a.diff - b.diff); const closest = milestones[0]; const diffInMinutes = Math.max(1, Math.ceil(closest.diff / 60));
                    showTextToast(`暫停中，約 ${diffInMinutes} 分鐘後進入 ${closest.name} `, 4500);
                }
                updateBacklightAndNorthStar(currentCycle, true, true); updateUniverseBackground(); saveSubjects(); spawnBubbles(); 
            } else {
                showTextToast('計時繼續', 2500); updateBacklightAndNorthStar(currentCycle, true, false); updateUniverseBackground(); clearBubbles(); hideQuote(); renderDrawer(); 
            }
        }
